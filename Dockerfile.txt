# Use an official C compiler image
FROM gcc:latest

# Set the working directory inside the container
WORKDIR /app

# Copy your C code into the container
COPY voting_server.c .

# Compile the C code into an executable
RUN gcc voting_server.c -o voting_server

# Expose the port (Render will override this with its env variable, but it's good practice)
EXPOSE 8080

# Run the compiled binary when the container starts
CMD ["./voting_server"]
